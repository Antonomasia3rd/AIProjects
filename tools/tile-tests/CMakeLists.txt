cmake_minimum_required(VERSION 3.16)
project(DesktopStubTileTests LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

foreach(test TileTextLayoutTests TileTextTemplateTests)
    add_executable(${test} ../../DesktopStub/tools/${test}.cpp)
    if(MSVC)
        target_compile_options(${test} PRIVATE /W4 /EHsc)
    else()
        target_compile_options(${test} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME ${test} COMMAND ${test})
endforeach()

if(WIN32)
    add_executable(TileTextRenderTests ../../DesktopStub/tools/TileTextRenderTests.cpp)
    target_compile_definitions(TileTextRenderTests PRIVATE UNICODE _UNICODE)
    target_link_libraries(TileTextRenderTests PRIVATE gdiplus user32 gdi32)
    if(MINGW)
        target_link_options(TileTextRenderTests PRIVATE -municode)
    endif()
    add_test(NAME TileTextRenderTests COMMAND TileTextRenderTests ${CMAKE_CURRENT_BINARY_DIR}/tile-render-smoke)
endif()
